### 将所有的canvas保存为图片
**代码一（推荐）：基于toBlob实现**
```js
document.querySelectorAll('canvas').forEach((canvas, index) => {
    try {
        // 使用异步的 toBlob 方法，避免阻塞主线程和内存溢出
        canvas.toBlob((blob) => {
            // 如果生成失败（例如遇到跨域污染），blob 会为 null
            if (!blob) {
                console.warn(`第 ${index + 1} 个 canvas 生成失败（可能是跨域污染）`);
                return;
            }

            // 生成一个临时的本地内存 URL，完美避开 Base64 的体积膨胀问题
            const url = URL.createObjectURL(blob);
            
            // 创建隐藏的 <a> 标签触发下载
            const link = document.createElement('a');
            link.download = `canvas-image-${index + 1}.png`;
            link.href = url;
            document.body.appendChild(link);
            link.click();
            
            // 清理 DOM 和释放内存
            document.body.removeChild(link);
            URL.revokeObjectURL(url); 
        }, 'image/png'); // 默认导出为 PNG 格式
    } catch (error) {
        console.error(`第 ${index + 1} 个 canvas 导出失败:`, error);
    }
});

```

**代码二（基于toDataURL）实现**
```js
document.querySelectorAll('canvas').forEach((canvas, index) => {
    try {
        // 将 canvas 内容转换为 base64 格式的 PNG 图片
        const dataURL = canvas.toDataURL('image/png');
        
        // 创建一个隐藏的 <a> 标签来触发下载
        const link = document.createElement('a');
        link.download = `canvas-image-${index + 1}.png`; // 设置下载的文件名
        link.href = dataURL;
        document.body.appendChild(link);
        link.click(); // 触发点击下载
        document.body.removeChild(link); // 清理 DOM
    } catch (error) {
        console.error(`第 ${index + 1} 个 canvas 保存失败:`, error);
    }
});
```


### 保存图片
```js
document.querySelectorAll('canvas').forEach((canvas, index) => {
    try {
        canvas.toBlob(function(blob) {
            if (!blob) {
                console.warn(`Canvas #${index} 无法导出，可能已被跨域污染。`);
                return;
            }
            const link = document.createElement('a');
            link.href = URL.createObjectURL(blob);
            link.download = `canvas_${index + 1}.png`; // 文件名会自动编号
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
            URL.revokeObjectURL(link.href); // 释放内存
        }, 'image/png');
    } catch (e) {
        console.error(`Canvas #${index} 导出失败:`, e);
    }
});

```
